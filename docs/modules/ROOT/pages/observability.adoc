== Diagnostics — ActivitySource

[source,csharp]
----
using Footing.Framework.Diagnostics;

using var a1 = FootingActivitySource.StartSqlTemplateRender("SELECT * FROM Users {WHERE} ...");
using var a2 = FootingActivitySource.StartEventBusPublish("UserCreatedEvent");
using var a3 = FootingActivitySource.StartSqlBatch("Users", 100);
using var a4 = FootingActivitySource.StartMigration("V1__create.sql");

// ActivitySource("Footing.Framework", "1.0") — 0 overhead sem listener
----

Configure OTel:

[source,csharp]
----
services.AddOpenTelemetry()
  .WithTracing(b => b
    .AddSource("Footing.Framework")
    .AddAspNetCoreInstrumentation()
    .AddOtlpExporter(o => o.Endpoint = new Uri("http://otel-collector:4317")));

// Visualize no Grafana/Jaeger — tags: db.statement, event.type, batch.size
----

[source,text]
----
  SqlTemplate.Render  ──> Activity SqlTemplateRender ──┐
  EventBus.Publish   ──> Activity EventBusPublish    ──┼──> OTel Collector ──> Grafana / Jaeger
  SqlBatch           ──> Activity SqlBatch           ──┘
----

== Health — /health

[source,csharp]
----
services.AddFootingHealthChecks(); // registra DbHealthCheck + EventBusHealthCheck
app.MapHealthChecks("/health");

// GET /health → {"status":"Healthy","results":{"db":"Healthy","eventbus":"Healthy"}}
----

Implementação:

[source,csharp]
----
public sealed class DbHealthCheck(IDbConnectionFactory f) : IHealthCheck {
  public async Task<HealthCheckResult> CheckHealthAsync(HealthCheckContext ctx, CancellationToken ct = default) {
    using var conn = f.CreateConnection();
    await ((System.Data.Common.DbConnection)conn).OpenAsync(ct);
    return HealthCheckResult.Healthy("db reachable");
  }
}
public sealed class EventBusHealthCheck(EventBus bus) : IHealthCheck {
  public Task<HealthCheckResult> CheckHealthAsync(HealthCheckContext ctx, CancellationToken ct = default)
    => Task.FromResult(bus.IsRunning ? HealthCheckResult.Healthy("eventbus running") : HealthCheckResult.Unhealthy("eventbus stopped"));
}
----

== Performance

Smoke tests validam `<2000ms` para `10k` renders:

[source,csharp]
----
[Fact] public void SqlTemplate_10k_under_2s() {
  var tpl = SqlTemplate.Parse("SELECT * FROM Users {WHERE} {IF:Name} AND Name=@Name {END} {ENDWHERE}");
  var sw = Stopwatch.StartNew();
  for(int i=0;i<10_000;i++) tpl.Render(new { Name="eder" });
  Assert.True(sw.ElapsedMilliseconds < 2000);
}
----

Para métricas formais, use `BenchmarkDotNet` (opcional):

== Transações DDL

Transações DDL agnósticas não são suportadas — `UseTransaction = true` lança `NotSupportedException`. Para atomicidade, use `BEGIN/COMMIT` no `.sql` quando o banco permite (ex. PostgreSQL). Ver xref:migrations.adoc#_transacoes_ddl[Migrations — Transações].
